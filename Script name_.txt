def search_monk_roster(db, keyword):
    keyword = (keyword or "").strip()
    if not keyword:
        return []

    return db.execute(
        """
        SELECT id, name, temple, invited, kudi, gift
        FROM roster
        WHERE instr(name, :keyword) > 0
        ORDER BY id
        """,
        {"keyword": keyword},
    ).fetchall()


# อ่านบัตรสำเร็จ ใช้เฉพาะชื่อตัวภาษาไทยจากบัตร
matches = search_monk_roster(db, card_data.get("given"))

# เจ้าหน้าที่ค้นเอง ใช้ข้อความจากช่อง "คำค้นในชื่อพระ"

ใช้ **ชื่อตัวภาษาไทย (`given_name`) จากบัตรประชาชนเป็นคำค้นอัตโนมัติ** 
ในช่อง **คำค้นในชื่อพระ** ค่ะ เช่น ถ้าบัตรอ่านได้ชื่อตัว `เลี่ยม` ให้ใส่ `เลี่ยม` ลงในช่องค้นหาแล้วค้นทันที 
โดยใช้เฉพาะชื่อตัวเป็นเงื่อนไขค้นหา

ให้ค้นในคอลัมน์ **`ชื่อพระ` (คอลัมน์ H ของ Excel)** แบบ **พบข้อความได้ทุกตำแหน่ง** เทียบเท่า `LIKE '%คำค้น%'` 

หากพบหลายรายการ ให้แสดงผลทั้งหมดพร้อมชื่อวัด เพื่อให้เจ้าหน้าที่ตรวจและเลือกรายการก่อนยืนยันลงทะเบียน 
หากอ่านบัตรไม่ได้หรือไม่มีชื่อตัว ให้เจ้าหน้าที่พิมพ์คำค้นเองได้
